cmake_minimum_required(VERSION 3.16)
project(PiTunesNative VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_AUTOMOC ON)
set(CMAKE_AUTORCC ON)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
find_package(Qt5 5.15 REQUIRED COMPONENTS Widgets Network Concurrent Svg)
include(CTest)
add_library(pitunes-core
  native/models/Domain.cpp native/models/LibraryModel.cpp
  native/networking/ApiClient.cpp native/artwork/ArtworkCache.cpp
  native/coverflow/CoverFlowGeometry.cpp native/coverflow/CoverFlowWidget.cpp)
target_include_directories(pitunes-core PUBLIC native)
target_link_libraries(pitunes-core PUBLIC Qt5::Widgets Qt5::Network Qt5::Concurrent Qt5::Svg)
target_compile_options(pitunes-core PRIVATE -Wall -Wextra -Wpedantic)
if(CMAKE_SIZEOF_VOID_P EQUAL 4 AND CMAKE_SYSTEM_PROCESSOR MATCHES "^(i[3-6]86|x86_64|amd64)$")
  # Debian 12's non-SSE2 Qt baseline is retained; never inherit the build host CPU.
  target_compile_options(pitunes-core PUBLIC -march=i686 -mtune=generic -mfpmath=387)
endif()
set(PITUNES_UI_SOURCES native/ui/MainWindow.cpp native/ui/PlayerIcons.cpp native/ui/SettingsDialog.cpp native/ui/TouchKeyboard.cpp native/ui/TrackDrawer.cpp)
add_executable(pitunes-native native/app/main.cpp ${PITUNES_UI_SOURCES} native/assets.qrc)
target_link_libraries(pitunes-native PRIVATE pitunes-core)
target_compile_options(pitunes-native PRIVATE -Wall -Wextra -Wpedantic)
install(TARGETS pitunes-native RUNTIME DESTINATION bin)
if(BUILD_TESTING)
  find_package(Qt5 5.15 REQUIRED COMPONENTS Test)
  add_executable(native-tests tests/native/NativeTests.cpp ${PITUNES_UI_SOURCES} native/assets.qrc)
  target_link_libraries(native-tests PRIVATE pitunes-core Qt5::Test)
  add_test(NAME native-contracts COMMAND native-tests)
  set_tests_properties(native-contracts PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen")
endif()
set(CPACK_GENERATOR DEB)
set(CPACK_PACKAGE_NAME pitunes-native)
set(CPACK_PACKAGING_INSTALL_PREFIX "/usr")
set(CPACK_PACKAGE_CONTACT "BASF2HD")
set(CPACK_PACKAGE_DESCRIPTION_SUMMARY "PiTunes native Qt Widgets appliance interface")
set(CPACK_DEBIAN_PACKAGE_SHLIBDEPS ON)
include(CPack)
